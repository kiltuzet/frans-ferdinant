-- Создание базы данных
CREATE DATABASE IF NOT EXISTS bus_depot;
USE bus_depot;

-- Создание таблицы типов пользователей (справочник)
CREATE TABLE IF NOT EXISTS user_types (
    id INT AUTO_INCREMENT PRIMARY KEY,
    type_name VARCHAR(50) NOT NULL UNIQUE,
    description VARCHAR(255)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы пользователей
CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    login VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    id_type_user INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_type_user) REFERENCES user_types(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы водителей
CREATE TABLE IF NOT EXISTS drivers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    license_number VARCHAR(20) NOT NULL UNIQUE,
    employee_number VARCHAR(20) NOT NULL UNIQUE,
    birth_date DATE,
    hire_date DATE NOT NULL,
    status ENUM('active', 'vacation', 'sick_leave', 'fired') DEFAULT 'active'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы автобусов
CREATE TABLE IF NOT EXISTS buses (
    id INT AUTO_INCREMENT PRIMARY KEY,
    mark VARCHAR(50) NOT NULL,
    model VARCHAR(50) NOT NULL,
    year INT,
    capacity INT NOT NULL,
    id_driver INT,
    registration_number VARCHAR(20) NOT NULL UNIQUE,
    status ENUM('active', 'maintenance', 'decommissioned') DEFAULT 'active',
    FOREIGN KEY (id_driver) REFERENCES drivers(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы маршрутов
CREATE TABLE IF NOT EXISTS routes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    route_number VARCHAR(10) NOT NULL UNIQUE,
    route_name VARCHAR(100) NOT NULL,
    description TEXT,
    distance_km DECIMAL(6,2),
    estimated_time_min INT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы остановок
CREATE TABLE IF NOT EXISTS stops (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    is_terminal BOOLEAN NOT NULL DEFAULT FALSE,
    address VARCHAR(255) NOT NULL,
    latitude DECIMAL(10,8),
    longitude DECIMAL(11,8)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы связи маршрутов и остановок
CREATE TABLE IF NOT EXISTS route_stops (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_route INT NOT NULL,
    id_stop INT NOT NULL,
    stop_order INT NOT NULL,
    arrival_time TIME,
    departure_time TIME,
    UNIQUE KEY (id_route, id_stop),
    FOREIGN KEY (id_route) REFERENCES routes(id) ON DELETE CASCADE,
    FOREIGN KEY (id_stop) REFERENCES stops(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы техосмотров
CREATE TABLE IF NOT EXISTS inspections (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_bus INT NOT NULL,
    inspection_date DATE NOT NULL,
    next_inspection_date DATE NOT NULL,
    result BOOLEAN NOT NULL,
    inspector_name VARCHAR(100),
    comments TEXT,
    FOREIGN KEY (id_bus) REFERENCES buses(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы пассажиров
CREATE TABLE IF NOT EXISTS passengers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_route INT,
    boarding_time DATETIME,
    disembarkation_time DATETIME,
    ticket_number VARCHAR(20),
    fare DECIMAL(6,2),
    FOREIGN KEY (id_route) REFERENCES routes(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Создание таблицы назначений автобусов на маршруты
CREATE TABLE IF NOT EXISTS bus_assignments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_bus INT NOT NULL,
    id_route INT NOT NULL,
    assignment_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    status ENUM('scheduled', 'in_progress', 'completed', 'cancelled') DEFAULT 'scheduled',
    FOREIGN KEY (id_bus) REFERENCES buses(id) ON DELETE CASCADE,
    FOREIGN KEY (id_route) REFERENCES routes(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
